Endpoints

Semua endpoint (kecuali yang ditandai Public) memerlukan otentikasi. Endpoint dengan label Session memerlukan login cookie, sedangkan label Permission memerlukan API Key yang sesuai.

JavaScript (Fetch)

const BASE_URL = "https://YOUR-AYAKA-DOMAIN.vercel.app";
const API_KEY = "ayaka_live_YOUR_SECRET_KEY";

async function sendMagicLink(email) {
  const res = await fetch(`${BASE_URL}/api/am/send`, {
    method: "POST",
    headers: { 
      "Content-Type": "application/json", 
      "X-API-Key": API_KEY 
    },
    body: JSON.stringify({ email, plan: "premium" }),
  });
  
  const result = await res.json();
  if (!res.ok) throw new Error(result.message || "Request failed");
  return result;
}

async function checkPremium(email) {
  const res = await fetch(
    `${BASE_URL}/api/am/status?email=${encodeURIComponent(email)}`,
    { headers: { "X-API-Key": API_KEY } }
  );
  return res.json();
}

PHP (cURL)

<?php
$baseUrl = "https://YOUR-AYAKA-DOMAIN.vercel.app";
$apiKey  = "ayaka_live_YOUR_SECRET_KEY";

function ayakaPost($path, $payload, $apiKey, $baseUrl) {
    $ch = curl_init($baseUrl . $path);
    curl_setopt_array($ch, [
        CURLOPT_RETURNTRANSFER => true,
        CURLOPT_POST           => true,
        CURLOPT_HTTPHEADER     => [
            "Content-Type: application/json",
            "X-API-Key: " . $apiKey,
        ],
        CURLOPT_POSTFIELDS => json_encode($payload),
        CURLOPT_TIMEOUT    => 20,
    ]);
    $response = curl_exec($ch);
    $status   = curl_getinfo($ch, CURLINFO_HTTP_CODE);
    if ($response === false) {
        throw new RuntimeException(curl_error($ch));
    }
    curl_close($ch);
    return ["status" => $status, "body" => json_decode($response, true)];
}

$result = ayakaPost("/api/am/send",
    ["email" => "user@example.com", "plan" => "premium"],
    $apiKey, $baseUrl);

echo "HTTP Status: " . $result["status"] . PHP_EOL;
echo json_encode($result["body"], JSON_PRETTY_PRINT);

Python (requests)

import requests

BASE_URL = "https://YOUR-AYAKA-DOMAIN.vercel.app"
API_KEY  = "ayaka_live_YOUR_SECRET_KEY"
HEADERS  = {"Content-Type": "application/json", "X-API-Key": API_KEY}

# Kirim Magic Link
r = requests.post(
    f"{BASE_URL}/api/am/send",
    headers=HEADERS,
    json={"email": "user@example.com", "plan": "premium"},
    timeout=20,
)
print(r.status_code, r.json())

# Cek status Premium
r = requests.get(
    f"{BASE_URL}/api/am/status",
    headers={"X-API-Key": API_KEY},
    params={"email": "user@example.com"},
    timeout=20,
)
print(r.status_code, r.json())

# Perpanjang Premium
r = requests.post(
    f"{BASE_URL}/api/am/renew",
    headers=HEADERS,
    json={"email": "user@example.com", "duration_days": 30},
    timeout=20,
)
print(r.status_code, r.json())
